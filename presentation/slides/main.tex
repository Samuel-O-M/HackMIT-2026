% conmed — the deck.
%
% Design intent: each slide is a single external figure built in its own folder
% and compiled to a backgroundless PDF. Beamer contributes almost nothing but
% the ground colour and the sequence — all layout lives in the figures, so what
% you compile in a figure folder is exactly what appears on the slide.
%
% Build from presentation/:  make deck
\documentclass[aspectratio=169]{beamer}
\input{shared/base}

\beamertemplatenavigationsymbolsempty
\setbeamertemplate{headline}{}
\setbeamertemplate{footline}{}
\setbeamersize{text margin left=0mm,text margin right=0mm}
\setbeamercolor{background canvas}{bg=ground}
\setbeamercolor{normal text}{fg=ink}
\usefonttheme{professionalfonts}

% one full-bleed figure per slide
\newcommand{\slidefig}[1]{%
  \begin{frame}[plain]
    \begin{tikzpicture}[remember picture,overlay]
      \node[inner sep=0pt,anchor=center] at (current page.center)
        {\includegraphics[width=\paperwidth,height=\paperheight]{#1}};
    \end{tikzpicture}
  \end{frame}}
