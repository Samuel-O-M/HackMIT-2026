% Slide 10 — privacy.
%
% The swap is the slide: every hosted call struck out, an open model in its
% place. Drawn rather than described, because "we can run locally" is a
% sentence and a crossed-out vendor is an argument.
%
% All three now have a local replacement — speech in, speech out, and the
% reasoning model — so the claim is the whole call rather than only the audio.
% Gemma was chosen for native function calling: the brain is tool-driven, and
% a model that cannot call a tool cannot stand in for the one that does.
%
% No filenames: a store is named by what it holds.

% A hosted service, struck through, its mark still legible behind the rule.
\newcommand{\struck}[4]{% #1 x  #2 y  #3 label  #4 mark
  \fill[surfii, rounded corners=2pt] (#1,#2) rectangle (#1+34,#2+11);
  \draw[line, line width=0.4pt, rounded corners=2pt] (#1,#2) rectangle (#1+34,#2+11);
  \node[anchor=center] at (#1+8,#2+5.5) {\includegraphics[height=3.4mm]{shared/marks/#4}};
  \node[anchor=west, font=\fontsize{6.2}{7.5}\selectfont\sffamily\color{inkiii}] at (#1+14,#2+5.5) {#3};
  \draw[alarm, line width=0.9pt] (#1+12.5,#2+5.5) -- (#1+31,#2+5.5);
}

% The open model that replaces it.
\newcommand{\localmodel}[4]{% #1 x  #2 y  #3 model  #4 provenance
  \fill[brand!8!white, rounded corners=2pt] (#1,#2) rectangle (#1+46,#2+11);
  \draw[brand, line width=0.6pt, rounded corners=2pt] (#1,#2) rectangle (#1+46,#2+11);
  \node[anchor=west, font=\fontsize{7.2}{8.6}\selectfont\sffamily\bfseries\color{ink}] at (#1+4,#2+7.2) {#3};
  \node[anchor=west, font=\fontsize{5.4}{6.5}\selectfont\sffamily\color{inkii}] at (#1+4,#2+3.2) {#4};
}
