% Slide 05 — the pipeline.
%
% Five stages, one wall: the agent proposes, a person decides, and the log is
% written only on Promote. That boundary is the whole design.
%
% Rewritten because the stage names were doing no work — "Ground", "Review"
% and "Promote" are only obvious once you already know the system. Each now
% says what actually happens in it.
%
% The two stores are drawn as cylinders rather than rectangles, and named for
% what they are rather than by path. A filename tells a judge nothing; the
% shape and the tables inside tell them it is a database and what it holds.

% A database: body, lit top ellipse, curved base, one band.
% #1 centre x  #2 base y  #3 radius x  #4 height  #5 body fill
\newcommand{\dbshape}[5]{%
  \pgfmathsetmacro{\ry}{#3*0.30}%
  \fill[#5] (#1-#3,#2) rectangle (#1+#3,#2+#4);
  \fill[#5] (#1,#2) ellipse (#3 and \ry);
  \fill[surf] (#1,#2+#4) ellipse (#3 and \ry);
  \draw[line, line width=0.4pt] (#1-#3,#2) -- (#1-#3,#2+#4);
  \draw[line, line width=0.4pt] (#1+#3,#2) -- (#1+#3,#2+#4);
  \draw[line, line width=0.4pt] (#1,#2+#4) ellipse (#3 and \ry);
  \draw[line, line width=0.4pt] (#1-#3,#2) arc[start angle=180, end angle=360, x radius=#3, y radius=\ry];
  \draw[lineii, line width=0.3pt] (#1-#3,#2+#4*0.55) arc[start angle=180, end angle=360, x radius=#3, y radius=\ry];
}
